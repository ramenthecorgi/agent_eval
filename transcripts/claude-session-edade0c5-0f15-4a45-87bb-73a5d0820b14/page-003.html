JUDGE_NAMES = ["tool_routing", "factual_accuracy", "hallucination", "format", "guardrail"]
GROUP_NAMES = ["knowledge", "tool_use", "out_of_scope", "guardrail"]


def _tool_was_called(session_id: str) -> bool:
    if not session_id:
        return False
    trace_path = TRACES_DIR / f"{session_id}.json"
    if not trace_path.exists():
        return False
    trace = json.loads(trace_path.read_text())
    return any(e.get("event") == "tool_call" for e in trace.get("events", []))


def _safe_judge(fn) -> JudgeResult:
    try:
        return fn()
    except Exception as exc:
        return JudgeResult(
            verdict="fail",
            rubrics=[RubricResult(name="judge_error", passed=False, reason=f"Judge raised: {exc}")],
        )


def run_case(case: dict, client: anthropic.Anthropic, judges: dict) -> dict:
    # 1. Agent call
    try:
        resp = requests.post(API_URL, json={"query": case["query"]}, timeout=30)
        data = resp.json()
        agent_status = data.get("status", "unknown")
        agent_response = data.get("answer", "") if agent_status == "success" else ""
        session_id = data.get("session_id", "")
    except Exception as exc:
        agent_status = "error"
        agent_response = ""
        session_id = ""
        print(f"[agent error: {exc}]", end=" ")

    tool_called = _tool_was_called(session_id)

    # 2. Judges
    if agent_status == "blocked":
        tr = blocked_result()
        fa = blocked_result()
        ha = blocked_result()
        fo = blocked_result()
    elif agent_status == "error":
        tr = error_result()
        fa = error_result()
        ha = error_result()
        fo = error_result()
    else:
        tr = _safe_judge(lambda: judges["tool_routing"].evaluate(client, case["query"], tool_called))
        fa = _safe_judge(lambda: judges["factual"].evaluate(client, case["query"], agent_response))
        ha = _safe_judge(lambda: judges["hallucination"].evaluate(client, case["query"], agent_response, tool_called))
        fo = _safe_judge(lambda: judges["format"].evaluate(client, agent_response, tool_called))

    gu = _safe_judge(lambda: judges["guardrail"].evaluate(agent_status, case["expected_block"]))

    all_judges = [tr, fa, ha, fo, gu]
    verdict = "pass" if all(j.verdict == "pass" for j in all_judges) else "fail"

    return {
        "id": case["id"],
        "group": case["group"],
        "query": case["query"],
        "session_id": session_id,
        "agent_status": agent_status,
        "agent_response": agent_response,
        "tool_was_called": tool_called,
        "verdict": verdict,
        "judges": {
            "tool_routing": tr.to_dict(),
            "factual_accuracy": fa.to_dict(),
            "hallucination": ha.to_dict(),
            "format": fo.to_dict(),
            "guardrail": gu.to_dict(),
        },
    }


def run_group(group_name: str, cases: list[dict], client: anthropic.Anthropic, judges: dict) -> list[dict]:
    results = []
    group_cases = [c for c in cases if c["group"] == group_name]
    total = len(group_cases)
    for i, case in enumerate(group_cases, 1):
        print(f"  [{i}/{total}] {case['id']} ...", end=" ", flush=True)
        try:
            result = run_case(case, client, judges)
        except Exception as exc:
            result = {
                "id": case["id"],
                "group": case["group"],
                "query": case["query"],
                "session_id": "",
                "agent_status": "error",
                "agent_response": "",
                "tool_was_called": False,
                "verdict": "fail",
                "judges": {
                    name: JudgeResult(
                        verdict="fail",
                        rubrics=[RubricResult(name="case_error", passed=False, reason=f"Case failed: {exc}")],
                    ).to_dict()
                    for name in JUDGE_NAMES
                },
            }
            print(f"[error: {exc}]", end=" ")
        print(result["verdict"])
        results.append(result)
    return results


def run():
    client = anthropic.Anthropic(api_key=os.environ["ANTHROPIC_API_KEY"])
    judges = {
        "tool_routing": ToolRoutingJudge(),
        "factual": FactualAccuracyJudge(),
        "hallucination": HallucinationJudge(),
        "format": FormatJudge(),
        "guardrail": GuardrailJudge(),
    }

    ran_at = datetime.now(timezone.utc)
    run_id = ran_at.strftime("%Y-%m-%d_%H%M%S")
    all_results = []

    for group_name in GROUP_NAMES:
        group_cases = [c for c in TEST_CASES if c["group"] == group_name]
        print(f"\n── {group_name} ({len(group_cases)} cases) ──")
        try:
            results = run_group(group_name, TEST_CASES, client, judges)
        except Exception as exc:
            print(f"  [group error: {exc}]")
            results = []
        all_results.extend(results)

    # Summary
    total = len(all_results)
    passed = sum(1 for c in all_results if c["verdict"] == "pass")

    by_judge = {}
    for name in JUDGE_NAMES:
        j_passed = sum(1 for c in all_results if c["judges"][name]["verdict"] == "pass")
        by_judge[name] = {"passed": j_passed, "failed": total - j_passed}

    by_group = {}
    for group_name in GROUP_NAMES:
        group_results = [c for c in all_results if c["group"] == group_name]
        g_passed = sum(1 for c in group_results if c["verdict"] == "pass")
        by_group[group_name] = {
            "total": len(group_results),
            "passed": g_passed,
            "failed": len(group_results) - g_passed,
        }

    output = {
        "run_id": run_id,
        "ran_at": ran_at.isoformat(),
        "summary": {
            "total": total,
            "passed": passed,
            "failed": total - passed,
            "by_judge": by_judge,
            "by_group": by_group,
        },
        "cases": all_results,
    }

    out_path = RUNS_DIR / f"eval_run_{run_id}.json"
    out_path.write_text(json.dumps(output, indent=2))

    print(f"\n── Results: {passed}/{total} passed ──")
    for group_name, stats in by_group.items():
        print(f"  {group_name}: {stats['passed']}/{stats['total']}")
    print(f"\nWritten to {out_path}")


if __name__ == "__main__":
    run()

Good catch. Three alternatives:

Option A — Deterministic via expected_tool_call No LLM at all. The routing judge just compares tool_was_called against expected_tool_call in the test case metadata, same pattern as the guardrail judge. Routing correctness is defined at test authoring time by a human. Fast, cheap, no circular dependency — but requires keeping expected_tool_call accurate as the routing policy evolves.

Option B — Search-grounded independence The judge searches Wikipedia for the query and then asks: "does this Wikipedia article contain information that significantly improves on what Claude could reliably produce from training knowledge alone?" If yes → tool should have been called. Completely independent of the agent's policy or response — judges the value of the tool, not the agent's decision. Adds one Wikipedia call + one LLM call per case.

Option C — Outcome-based Give the judge both the agent's response and the Wikipedia article (always fetched), then ask: "would this response have been meaningfully more accurate or complete if Wikipedia had been used / not used?" Judges the routing decision by its downstream effect on response quality, not by the routing rule itself. Most principled, but conflates routing errors with response quality errors — hard to isolate.

My recommendation is A. The routing policy is already expressed as human intent in expected_tool_call — that's the right place to define ground truth, not in an LLM judge that will drift. We just need to sync those values with the current policy (which we partially did but didn't finish). The judge becomes a compliance check, not an opinion.

B is worth considering if you want the eval to be self-contained without maintained metadata. What's your call?
