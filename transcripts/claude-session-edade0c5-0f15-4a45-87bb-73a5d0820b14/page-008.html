    function renderCaseCard(c, idx) {
      const verdictCls = c.verdict === 'pass' ? 'verdict-pass' : 'verdict-fail';
      const badge = `<span class="badge ${c.verdict === 'pass' ? 'badge-pass' : 'badge-fail'}">${c.verdict.toUpperCase()}</span>`;
      const failedJudges = Object.entries(c.judges)
        .filter(([, j]) => j.verdict === 'fail')
        .map(([name]) => JUDGE_LABELS[name] || name)
        .join(', ');
      const failureHint = failedJudges ? `<span class="case-failures">${failedJudges} ✗</span>` : '';
      const queryShort = c.query.length > 70 ? c.query.slice(0, 70) + '…' : c.query;

      // Only render judges present in this case's data
      const judgeSections = Object.entries(c.judges).map(([key, j]) => {
        const label = JUDGE_LABELS[key] || key;
        const jBadge = `<span class="badge ${j.verdict === 'pass' ? 'badge-pass' : 'badge-fail'}">${j.verdict.toUpperCase()}</span>`;
        const rubrics = j.rubrics.map(r => `
          <div class="rubric-row">
            <span class="rubric-name">${r.name}</span>
            <span class="rubric-chip ${r.passed ? 'chip-pass' : 'chip-fail'}">${r.passed ? 'PASS' : 'FAIL'}</span>
            <span class="rubric-reason">${escapeHtml(r.reason)}</span>
          </div>`).join('');
        return `
          <div class="judge-section">
            <div class="judge-section-header">${label} ${jBadge}</div>
            ${rubrics}
          </div>`;
      }).join('');

      const hasResponse = c.agent_response && c.agent_response.length > 0;
      const responseBlock = hasResponse ? `
        <span class="agent-response-toggle" onclick="toggleAgentResponse(${idx})">▶ agent response</span>
        <div class="agent-response-box" id="ar-${idx}">${escapeHtml(c.agent_response)}</div>` : '';

      return `
        <div class="case-card ${verdictCls}">
          <div class="case-header" onclick="toggleCase(${idx})">
            <span class="case-id">${c.id}</span>
            <span class="case-query">${escapeHtml(queryShort)}</span>
            ${failureHint}
            ${badge}
          </div>
          <div class="case-body" id="case-${idx}">
            ${responseBlock}
            ${judgeSections}
          </div>
        </div>`;
    }

    function toggleCase(idx) {

    }
