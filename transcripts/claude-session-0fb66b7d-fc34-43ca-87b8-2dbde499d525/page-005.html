170	        self.traces_dir.mkdir(parents=True, exist_ok=True)
171	        session = {

173	            "started_at": self.started_at,
174	            "user_query": self.user_query,
175	            "events": self.events,
176	            "status": status,
177	            "total_duration_ms": int((time.monotonic() - self._start) * 1000),
178	            "total_usage": total_usage,
179	            "final_answer": final_answer,
180	            "completed_at": datetime.now(timezone.utc).isoformat(),

392	from dataclasses import dataclass, field
393	from typing import Protocol
394	
395	_DEFAULT_BLOCKED_KEYWORDS = ["ignore all previous", "jailbreak", "system prompt", "ignore previous instructions"]
396	_DEFAULT_MAX_LENGTH = 500
397	
398	
399	@dataclass
400	class GuardrailResult:
401	    passed: bool
402	    reason: str | None
403	    rules_evaluated: list[dict]
404	    metadata: dict = field(default_factory=dict)
405	
406	
407	class Guardrail(Protocol):
408	    def check_request(self, text: str) -> GuardrailResult: ...
409	    def check_response(self, text: str) -> GuardrailResult: ...
410	
411	
412	class RuleBasedGuardrail:
413	    def __init__(
414	        self,
415	        max_length: int = _DEFAULT_MAX_LENGTH,
416	        blocked_keywords: list[str] | None = None,
417	    ):
418	        self.max_length = max_length
419	        self.blocked_keywords = blocked_keywords if blocked_keywords is not None else _DEFAULT_BLOCKED_KEYWORDS
420	
421	    def _evaluate(self, text: str) -> GuardrailResult:
422	        rules: list[dict] = []
423	        blocked_by: str | None = None
424	        text_lower = text.lower()
425	
426	        length_ok = len(text) <= self.max_length
427	        rules.append({"rule": "max_length", "passed": length_ok})
428	        if not length_ok and blocked_by is None:
429	            blocked_by = "max_length"
430	
431	        keywords_ok = not any(kw in text_lower for kw in self.blocked_keywords)
432	        rules.append({"rule": "blocked_keywords", "passed": keywords_ok})
433	        if not keywords_ok and blocked_by is None:
434	            blocked_by = "blocked_keywords"
435	
436	        return GuardrailResult(passed=blocked_by is None, reason=blocked_by, rules_evaluated=rules)
437	
438	    def check_request(self, text: str) -> GuardrailResult:
439	        return self._evaluate(text)
440	
441	    def check_response(self, text: str) -> GuardrailResult:
442	        return self._evaluate(text)
443	```

596	_TOOL_DEFINITION = {
597	    "name": "search_wikipedia",
598	    "description": "Search Wikipedia for information about a topic and return a summary.",
599	    "input_schema": {
600	        "type": "object",
601	        "properties": {
602	            "query": {"type": "string", "description": "The search term to look up on Wikipedia."}
603	        },
604	        "required": ["query"],
605	    },
606	}
607	
608	
609	def _serialize_block(block) -> dict:
610	    if block.type == "text":
611	        return {"type": "text", "text": block.text}
612	    if block.type == "tool_use":
613	        return {"type": "tool_use", "id": block.id, "name": block.name, "input": block.input}
614	    return {"type": block.type}
615	
616	
617	def run_agent(query: str, tracer: Tracer, client: anthropic.Anthropic) -> tuple[str, dict]:
618	    messages: list[dict] = [{"role": "user", "content": query}]
619	    total_input = 0
620	    total_output = 0
621	    turn = 0
622	
623	    while True:
624	        turn += 1
625	        tracer.emit({"event": "llm_request", "turn": turn, "messages": messages})
626	
627	        t0 = time.monotonic()
628	        response = client.messages.create(
629	            model=_MODEL,
630	            max_tokens=_MAX_TOKENS,
631	            tools=[_TOOL_DEFINITION],
632	            messages=messages,
633	        )
634	        duration_ms = int((time.monotonic() - t0) * 1000)
635	        total_input += response.usage.input_tokens
636	        total_output += response.usage.output_tokens
637	
638	        tracer.emit({
639	            "event": "llm_response",
640	            "turn": turn,
641	            "duration_ms": duration_ms,
642	            "stop_reason": response.stop_reason,
643	            "content": [_serialize_block(b) for b in response.content],
644	            "usage": {"input_tokens": response.usage.input_tokens, "output_tokens": response.usage.output_tokens},
645	        })
646	
647	        if response.stop_reason == "end_turn":
648	            text = next((b.text for b in response.content if b.type == "text"), "")
649	            return text, {"input_tokens": total_input, "output_tokens": total_output}
650	
651	        if response.stop_reason == "tool_use":
652	            messages.append({"role": "assistant", "content": [_serialize_block(b) for b in response.content]})
653	            tool_results = []
654	            for block in response.content:
655	                if block.type != "tool_use":
656	                    continue
657	                tracer.emit({"event": "tool_call", "turn": turn, "tool": block.name, "input": block.input})
658	                t1 = time.monotonic()
659	                try:
660	                    result = search_wikipedia(block.input["query"])
661	                    tool_duration_ms = int((time.monotonic() - t1) * 1000)
662	                    tracer.emit({
663	                        "event": "tool_result",
664	                        "turn": turn,
665	                        "tool": block.name,
666	                        "duration_ms": tool_duration_ms,
667	                        "output": result,
668	                        "output_length": len(result),
669	                    })
670	                    tool_results.append({"type": "tool_result", "tool_use_id": block.id, "content": result})
671	                except Exception as exc:
672	                    tool_duration_ms = int((time.monotonic() - t1) * 1000)
673	                    tracer.emit({"event": "error", "turn": turn, "source": "tool", "message": str(exc)})
674	                    tool_results.append({
675	                        "type": "tool_result", "tool_use_id": block.id,
676	                        "content": f"Error: {exc}", "is_error": True,
677	                    })
678	            messages.append({"role": "user", "content": tool_results})
679	        else:
680	            tracer.emit({"event": "error", "turn": turn, "source": "llm",
681	                         "message": f"Unexpected stop_reason: {response.stop_reason}"})
682	            raise ValueError(f"Unexpected stop_reason: {response.stop_reason}")

            "total_usage": total_usage,
            "final_answer": final_answer,
            "completed_at": datetime.now(timezone.utc).isoformat(),
        }
        (self.traces_dir / f"{self.session_id}.json").write_text(json.dumps(session, indent=2))

from typing import Protocol

_DEFAULT_BLOCKED_KEYWORDS = ["ignore all previous", "jailbreak", "system prompt", "ignore previous instructions"]

def _serialize_block(block) -> dict:
    if block.type == "text":
        return {"type": "text", "text": block.text}
    if block.type == "tool_use":
        return {"type": "tool_use", "id": block.id, "name": block.name, "input": block.input}
    return {"type": block.type}


def run_agent(query: str, tracer: Tracer, client: anthropic.Anthropic) -> tuple[str, dict]:
    messages: list[dict] = [{"role": "user", "content": query}]
    total_input = 0
    total_output = 0
    turn = 0

    while True:
        turn += 1
        tracer.emit({"event": "llm_request", "turn": turn, "messages": messages})

        t0 = time.monotonic()
        response = client.messages.create(
            model=_MODEL,
            max_tokens=_MAX_TOKENS,
            tools=[_TOOL_DEFINITION],
            messages=messages,
        )
        duration_ms = int((time.monotonic() - t0) * 1000)
        total_input += response.usage.input_tokens
        total_output += response.usage.output_tokens

        tracer.emit({
            "event": "llm_response",
            "turn": turn,
            "duration_ms": duration_ms,
            "stop_reason": response.stop_reason,
            "content": [_serialize_block(b) for b in response.content],
            "usage": {"input_tokens": response.usage.input_tokens, "output_tokens": response.usage.output_tokens},
        })

        if response.stop_reason == "end_turn":
            text = next((b.text for b in response.content if b.type == "text"), "")
            return text, {"input_tokens": total_input, "output_tokens": total_output}

        if response.stop_reason == "tool_use":
            messages.append({"role": "assistant", "content": [_serialize_block(b) for b in response.content]})
            tool_results = []
            for block in response.content:
                if block.type != "tool_use":
                    continue
                tracer.emit({"event": "tool_call", "turn": turn, "tool": block.name, "input": block.input})
                t1 = time.monotonic()
                try:
                    result = search_wikipedia(block.input["query"])
                    tool_duration_ms = int((time.monotonic() - t1) * 1000)
                    tracer.emit({
                        "event": "tool_result",
                        "turn": turn,
                        "tool": block.name,
                        "duration_ms": tool_duration_ms,
                        "output": result,
                        "output_length": len(result),
                    })
                    tool_results.append({"type": "tool_result", "tool_use_id": block.id, "content": result})
                except Exception as exc:
                    tool_duration_ms = int((time.monotonic() - t1) * 1000)
                    tracer.emit({"event": "error", "turn": turn, "source": "tool", "message": str(exc)})
                    tool_results.append({
                        "type": "tool_result", "tool_use_id": block.id,
                        "content": f"Error: {exc}", "is_error": True,
                    })
            messages.append({"role": "user", "content": tool_results})
        else:
            tracer.emit({"event": "error", "turn": turn, "source": "llm",

@app.post("/api/chat")
async def chat(request: ChatRequest):
    session_id = str(uuid.uuid4())
    tracer = Tracer(session_id=session_id, user_query=request.query, traces_dir=TRACES_DIR)
    total_usage: dict = {"input_tokens": 0, "output_tokens": 0}

    t0 = time.monotonic()
    req_result = guardrail.check_request(request.query)
    tracer.emit({
        "event": "request_safety_check",
        "duration_ms": int((time.monotonic() - t0) * 1000),
        "passed": req_result.passed,
        "rules_evaluated": req_result.rules_evaluated,
        "blocked_by": req_result.reason,
    })

    if not req_result.passed:
        tracer.save(status="blocked", final_answer=None, total_usage=total_usage)
        return JSONResponse(status_code=400, content={
            "session_id": session_id, "status": "blocked",
            "blocked_at": "request", "reason": req_result.reason,
        })

    try:
        answer, total_usage = run_agent(request.query, tracer, _get_client())
    except Exception as exc:
        tracer.emit({"event": "error", "source": "agent", "message": str(exc)})
        tracer.save(status="error", final_answer=None, total_usage=total_usage)
        raise HTTPException(status_code=500, detail=str(exc))

    t1 = time.monotonic()
    resp_result = guardrail.check_response(answer)
    tracer.emit({
        "event": "response_safety_check",
        "duration_ms": int((time.monotonic() - t1) * 1000),
        "passed": resp_result.passed,
        "rules_evaluated": resp_result.rules_evaluated,
        "blocked_by": resp_result.reason,
    })

    if not resp_result.passed:
        tracer.save(status="blocked", final_answer=None, total_usage=total_usage)
        return JSONResponse(status_code=400, content={
            "session_id": session_id, "status": "blocked",
            "blocked_at": "response", "reason": resp_result.reason,
        })

    tracer.emit({"event": "final_answer", "text": answer})
    tracer.save(status="success", final_answer=answer, total_usage=total_usage)
    return {"session_id": session_id, "answer": answer, "status": "success"}
